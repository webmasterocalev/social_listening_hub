async function processCSV(csvText) { // Assume que não tem cabeçalho, ou se tiver, precisa pular. // O formato esperado é: DATA, CANAL, USUARIO, LINK const rows = csvText.split('\n'); let toImport = []; // Preparar índice de duplicidade com dados atuais // Normaliza o link (trim) para garantir comparação justa const existingIndex = new Set(mentions.map(m => `${(m.user||'').toLowerCase()}|${(m.link||'').trim()}|Reação`)); // Função Robust para ler linha CSV (Respeita aspas "Value, with comma") const parseLine = (line) => { const result = []; let current = ''; let inQuotes = false; for (let i = 0; i < line.length; i++) { const char = line[i]; if (char === '"') { inQuotes = !inQuotes; } else if (char === ',' && !inQuotes) { result.push(current.trim()); current = ''; } else { current += char; } } result.push(current.trim()); return result; }; // Parsear CSV rows.forEach(row => { // Ignora linhas vazias if (!row.trim()) return; // Usa o parser robusto em vez de split simples const cols = parseLine(row); // Validação básica de colunas if (cols.length < 4) return; // Remove aspas extras que podem ter sobrado const clean = (str) => str ? str.replace(/^"|"$/g, '') : ''; const date = clean(cols[0]); const channel = clean(cols[1]); const user = clean(cols[2]).replace('@', ''); // Limpa @ se tiver const link = clean(cols[3]); // Se for cabeçalho (Data, Canal...), ignora if (date.toLowerCase().includes('data') || link.toLowerCase().includes('link')) return; // Se o link estiver vazio, pula (segurança) if (!link || link.length < 5) return; // Checa duplicidade (Agora com link garantido) const key = `${user.toLowerCase()}|${link}|Reação`; // Debug no console para você verificar se precisar (F12) // console.log('Processando:', key, 'Existe?', existingIndex.has(key)); if (!existingIndex.has(key)) { toImport.push({ date: date, channel: channel, user: user, link: link, type: 'Reação', sentiment: 'Positivo', topic: 'Engajamento', content: '(Reação / Like)', id: Date.now() + Math.random() // ID temporário }); existingIndex.add(key); // Adiciona ao índice para evitar duplicados no próprio CSV } }); if (toImport.length === 0) { showFeedback('Nenhuma curtida nova encontrada (ou erro de leitura).', 'error'); resetImportUI(); return; } // Enviar em Fila (Um por um para não sobrecarregar) importMsg.textContent = `Importando ${toImport.length} curtidas...`; for (let i = 0; i < toImport.length; i++) { const item = toImport[i]; try { await fetch(SCRIPT_URL, { method: 'POST', mode: 'no-cors', redirect: 'follow', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify(item) }); // Atualiza localmente mentions.unshift(item); // Atualiza barra const percent = Math.round(((i + 1) / toImport.length) * 100); progressBar.style.width = `${percent}%`; importCount.textContent = `${percent}%`; } catch (err) { console.error('Erro ao importar item', item, err); } } // Finalização updateDashboard(); populateChannelFilter(); showFeedback(`${toImport.length} curtidas importadas com sucesso!`); resetImportUI(); }
